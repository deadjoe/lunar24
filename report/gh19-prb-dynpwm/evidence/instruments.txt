"""Unit-level proof of the wiring's TWO SURFACES, against the gate's own code: parse a criteria whose
`code_cellset` is a sentinel, record one CELLSET refusal through the real `refuse()`, and require the
sentinel on the per-code detail line AND on the counts line. The pre-wiring copy is run through the same
script, where the same call must print the DEFAULT literal instead -- which is what makes this a
discrimination rather than a demonstration of the sentinel's existence."""
import hashlib, importlib.util, io, sys, contextlib

def load(path, name):
    spec = importlib.util.spec_from_file_location(name, path)
    m = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(m)
    return m

SENT = "LIVENESS-SENTINEL-7Q4Z"
src = "report/gh19-prb-dynpwm/acceptance_criteria.tsv"
crit = "/tmp/wire_unit_criteria.tsv"
open(crit, "w").write(open(src).read().replace("criterion\tcode_cellset\tCELLSET\n",
                                               "criterion\tcode_cellset\t%s\n" % SENT))
print("CRITERIA sentinel_applied=%s sha256=%s" % (SENT in open(crit).read(),
      hashlib.sha256(open(crit, "rb").read()).hexdigest()))

for tag, path in (("shipped", "tools/gh19_prb_dynpwm_acceptance.py"),
                  ("prewiring", "/tmp/gate_prewiring.py")):
    m = load(path, "gate_" + tag)
    buf = io.StringIO()
    with contextlib.redirect_stdout(buf):
        g = m.Gate()
        r = m.parse_criteria(g, crit)
        g.refuse("CELLSET", "unit-test refusal")
        m.finish(g, "REFUSE", m.EXIT_REFUSE)   # the gate's own constant; the return value is discarded here
    out = buf.getvalue()
    detail = [l for l in out.splitlines() if l.startswith("PRB-REFUSE-")]
    counts = [l for l in out.splitlines() if " COUNT " in l]
    print("--- %s ---" % tag)
    print("  codes_map=%s" % (g.codes,))
    print("  detail_lines=%r" % (detail,))
    print("  counts_lines=%r" % (counts,))
    print("  sentinel_on_detail=%s sentinel_on_counts=%s default_on_either=%s"
          % (any(SENT in l for l in detail), any(SENT in l for l in counts),
             any("CELLSET" in l for l in detail + counts)))

"""Per-key liveness: for EACH of the nine criteria `code_*` rows, rename it to a sentinel in a copy of
the criteria, parse that copy with the gate's own parser, record a refusal under the recorded suffix,
and require the sentinel on BOTH printed surfaces.

WHAT THIS DOES AND DOES NOT PROVE. It proves the mechanism is live for all nine: the map is consulted at
`refuse()`/`fail()` and the recorded string feeds both surfaces. It does NOT prove that a real run
reaches each code -- reachability is a property of the controls, not of the wiring, and is reported
separately. The map is keyed by the suffix DERIVED FROM THE KEY NAME (`code_missing_arm` -> `MISSING-ARM`),
which is also the value the call sites record -- including the one that records through a local variable
rather than a literal.
"""
import importlib.util, io, contextlib, re

SENT = "LIVENESS-SENTINEL-7Q4Z"
src = "report/gh19-prb-dynpwm/acceptance_criteria.tsv"
base = open(src, encoding="utf-8").read()
crit = "/tmp/wire_per_key_criteria.tsv"

spec = importlib.util.spec_from_file_location("gate", "tools/gh19_prb_dynpwm_acceptance.py")
m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m)

rows = [(f[1], f[2]) for f in (l.split("\t") for l in base.splitlines())
        if f[0] == "criterion" and f[1].startswith("code_")]
print("criteria code_* rows: %d" % len(rows))
print("%-24s %-18s %-6s %-6s %-6s" % ("key", "value", "detail", "counts", "default"))
all_ok = True
for key, val in rows:
    suffix = key[len("code_"):].replace("_", "-").upper()
    text = base.replace("criterion\t%s\t%s\n" % (key, val),
                        "criterion\t%s\t%s\n" % (key, SENT))
    assert SENT in text, key
    open(crit, "w", encoding="utf-8", newline="\n").write(text)
    buf = io.StringIO()
    with contextlib.redirect_stdout(buf):
        g = m.Gate()
        m.parse_criteria(g, crit)
        g.refuse(suffix, "per-key probe")
        m.finish(g, "REFUSE", m.EXIT_REFUSE)   # the gate's own constant; the return value is discarded here
    out = buf.getvalue()
    det = [l for l in out.splitlines() if l.startswith("PRB-REFUSE-")]
    cnt = [l for l in out.splitlines() if " COUNT " in l]
    d = any(SENT in l for l in det)
    c = any(SENT in l for l in cnt)
    dflt = any(val in l for l in det + cnt)
    ok = d and c and not dflt
    all_ok = all_ok and ok
    print("%-24s %-18s %-6s %-6s %-6s %s" % (key, val, d, c, dflt, "OK" if ok else "FAIL"))
print("ALL NINE LIVE ON BOTH SURFACES: %s" % all_ok)

"""What codes does the control suite EXERCISE? Read from the CONTROLS table itself rather than from a
promise: each entry names the token its mutation must produce. The gate's own output is not reprinted on
a HIT, so the declaration is the only per-control attribution available -- and it is the declaration the
suite is judged against, so a suite that fired a different code would MISS."""
import importlib.util, re
spec = importlib.util.spec_from_file_location("nc", "tools/gh19_prb_dynpwm_gate_negcontrol.py")
nc = importlib.util.module_from_spec(spec); spec.loader.exec_module(nc)

crit = open("report/gh19-prb-dynpwm/acceptance_criteria.tsv", encoding="utf-8").read()
named = {}
for ln in crit.splitlines():
    f = ln.split("\t")
    if f[0] == "criterion" and f[1].startswith("code_"):
        named[f[1][len("code_"):].replace("_", "-").upper()] = f[2]
src = open("tools/gh19_prb_dynpwm_acceptance.py", encoding="utf-8").read()
emitted = sorted(set(re.findall(r'g\.(?:refuse|fail)\("([^"]+)"', src)))
print("controls in the suite: %d" % len(nc.CONTROLS))
print("\ncontrol -> code it must produce (and codes it must NOT produce)")
for c in nc.CONTROLS:
    print("  %-30s %-28s forbidden=%s" % (c[0], c[3], ",".join(c[4]) or "-"))
exercised = sorted({c[3].split("PRB-REFUSE-")[-1].split("PRB-FAIL-")[-1] for c in nc.CONTROLS})
print("\ncodes the suite exercises: %s" % ", ".join(exercised))
print("\nreconciliation against the nine criteria-named suffixes:")
for s in sorted(named):
    print("  %-18s named=%-18s exercised_by_suite=%s" % (s, named[s], s in exercised))
print("\ncodes the gate can print that NO criteria row names (%d):" % len(set(emitted) - set(named)))
print("  %s" % ", ".join(sorted(set(emitted) - set(named))))
print("  of those, the suite exercises: %s"
      % (", ".join(sorted((set(emitted) - set(named)) & set(exercised))) or "(none)"))
