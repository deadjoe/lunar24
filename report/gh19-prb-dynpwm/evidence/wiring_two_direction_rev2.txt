"""A/B the code-wiring change -- REV 2, against the gate and criteria as they stand now.

Why a second revision exists: the 17:18 run paired a pre-wiring copy derived from the gate as it stood
at 16:51 with the gate as it stood at 17:18. The shipped gate moved again afterwards (the director's
ruling that the code vocabulary must be complete: seven more `code_*` suffixes named), so the old copy
no longer differs from the shipped gate by the wiring alone -- measured, it differs by 8 hunks. An
identity established against a superseded pair is not evidence about the artifact under review, so the
comparison is re-derived from the current tree rather than carried forward.

IDENTITY  -- both builds run the control suite EXCEPT the liveness control; the two transcripts must be
             byte-identical. That is the whole claim: the wiring cannot change a byte of this gate's
             output while the criteria holds the default values.
LIVENESS  -- both builds run ONLY the liveness control. The wired build must print the criteria's
             sentinel on both surfaces; the pre-wiring build must MISS and print its default literal.
The criteria is copied from the repo and the copy is asserted byte-identical before use, so neither
build is ever handed a file this script invented.
"""
import argparse, hashlib, os, shutil, subprocess, sys

def sha(p):
    return hashlib.sha256(open(p, "rb").read()).hexdigest()

ap = argparse.ArgumentParser()
ap.add_argument("--tmp", default="/tmp/prb-pipe")
ap.add_argument("--repo", default=".")
ap.add_argument("--work", default="/tmp/prb-ab-rev2")
ap.add_argument("--pre", default="/tmp/gate_prewiring_rev2.py",
                help="the pre-wiring gate build; must differ from --shipped by the wiring alone")
a = ap.parse_args()

repo = os.path.abspath(a.repo)
crit_src = os.path.join(repo, "report/gh19-prb-dynpwm/acceptance_criteria.tsv")
shipped = os.path.join(repo, "tools/gh19_prb_dynpwm_acceptance.py")
pre = a.pre
nc = os.path.join(repo, "tools/gh19_prb_dynpwm_gate_negcontrol.py")

os.makedirs(a.work, exist_ok=True)
crit = os.path.join(a.work, "criteria.tsv")
shutil.copyfile(crit_src, crit)
assert sha(crit) == sha(crit_src), "the sandbox copy must be the repo file byte for byte"
print("CRITERIA src_sha256=%s copy_sha256=%s" % (sha(crit_src), sha(crit)))
print("GATE shipped_sha256=%s" % sha(shipped))
print("GATE prewiring_sha256=%s path=%s" % (sha(pre), pre))

# The isolation this whole comparison rests on, asserted rather than assumed: the two builds must differ
# by the wiring hunk and nothing else. If they do not, the comparison is not about the wiring.
d = subprocess.run(["diff", "-u", pre, shipped], capture_output=True, text=True)
hunks = [l for l in d.stdout.splitlines() if l.startswith("@@")]
wired = [l[1:] for l in d.stdout.splitlines()
         if l.startswith("+") and not l.startswith("+++")]
stripped = [l[1:] for l in d.stdout.splitlines()
            if l.startswith("-") and not l.startswith("---")]
print("ISOLATION diff_hunks=%d wired_lines=%d removed_lines=%d" % (len(hunks), len(wired), len(stripped)))
for l in wired:
    print("ISOLATION   +%s" % l)
for l in stripped:
    print("ISOLATION   -%s" % l)
ok_isolation = (len(hunks) == 1 and len(wired) == 3 and len(stripped) == 1
                and stripped[0].strip().startswith("# PRE-WIRING CONTROL COPY"))
print("ISOLATION wiring_only=%s" % ok_isolation)

tmp = a.tmp
common = ["--criteria", crit,
          "--plan", os.path.join(tmp, "cand", "gh19_s3_plan.tsv"),
          "--cand-report", os.path.join(tmp, "cand_report.txt"),
          "--cand-arm", os.path.join(tmp, "cand"),
          "--neut-report", os.path.join(tmp, "neut_report.txt"),
          "--neut-arm", os.path.join(tmp, "neut"),
          "--degen-static-arm", os.path.join(tmp, "cand"),
          "--degen-dyn-arm", os.path.join(tmp, "degen")]

def run(gate, extra):
    p = subprocess.run([sys.executable, nc] + common + ["--gate", gate] + extra,
                       capture_output=True, text=True, encoding="utf-8")
    return p.returncode, p.stdout

out = {}
for tag, gate in (("shipped", shipped), ("prewiring", pre)):
    rc, txt = run(gate, ["--skip", "criteria_code_key_is_read"])
    f = os.path.join(a.work, "suite_%s.txt" % tag)
    open(f, "w").write(txt)
    out[("suite", tag)] = (rc, txt, f)
    print("SUITE %-10s exit=%d sha256=%s %s" % (tag, rc, sha(f), f))
    rc, txt = run(gate, ["--only", "criteria_code_key_is_read"])
    f = os.path.join(a.work, "liveness_%s.txt" % tag)
    open(f, "w").write(txt)
    out[("live", tag)] = (rc, txt, f)
    print("LIVE  %-10s exit=%d sha256=%s %s" % (tag, rc, sha(f), f))

s_sh, s_pre = out[("suite", "shipped")][1], out[("suite", "prewiring")][1]
print("IDENTITY suite_transcripts_identical=%s" % (s_sh == s_pre))
print("IDENTITY shipped_lines=%d prewiring_lines=%d" % (len(s_sh.splitlines()), len(s_pre.splitlines())))
# An identity of two empty or near-empty transcripts proves nothing, so the shape is printed too.
n_report = len([l for l in s_sh.splitlines() if "PRB REPORT" in l])
print("IDENTITY shipped_report_lines=%d shipped_ok_lines=%d"
      % (n_report, len([l for l in s_sh.splitlines() if l.startswith("NEGCTL-PRB ok=")])))
print("IDENTITY shipped_verdict=%s" % [l for l in s_sh.splitlines() if l.startswith("NEGCTL-PRB verdict=")])
if s_sh != s_pre:
    import difflib
    for ln in list(difflib.unified_diff(s_pre.splitlines(), s_sh.splitlines(),
                                        "prewiring", "shipped", lineterm=""))[:40]:
        print("  | " + ln)

for tag in ("shipped", "prewiring"):
    rc, txt, _ = out[("live", tag)]
    hit = [l for l in txt.splitlines() if l.startswith("CONTROL criteria_code_key_is_read")]
    print("LIVENESS %-10s exit=%d %s" % (tag, rc, hit[0].strip() if hit else "(no control line)"))
    print("           sentinel_in_transcript=%s default_literal_in_transcript=%s"
          % ("LIVENESS-SENTINEL-7Q4Z" in txt, "PRB-REFUSE-CELLSET" in txt))
